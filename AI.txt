Aim: Implement the Breadth First Search algorithm to solve a given problem.

Input:

from collections import deque
def bfs(graph, start):
    visited = []
    queue = deque([start])

    while queue:
        node = queue.popleft()
        if node not in visited:
            visited.append(node)
            queue.extend(graph[node])
    print("BFS Traversal:", visited)
graph = {
    'A' :['B','C'],
    'B' :['D','E'],
    'C' :['F'],
    'D' :[],
    'E' :['F'],
    'F' :[]
    }
bfs(graph,'A')

from collections import deque
def idfs(graph, start):
    visited = []
    stack = [start]

    while stack:
        node = stack.pop()
        if node not in visited:
            visited.append(node)
            stack.extend(reversed(graph[node]))
    print("IDFS Traversal:", visited)
graph = {
    'A' : ['B','C'],
    'B' : ['D','E'],
    'C' : ['F'],
    'D' : ['G', 'H'],
    'E' : ['F'],
    'F' : [],
    'G' : [],
    'H' : []
    }
idfs(graph,'A')



PRACTICAL NO.2(A)
Aim: Implement the Breadth First Search algorithm to solve a given problem.

Input:

import heapq
graph = {
    'A':[('B',1), ('C',3)],
    'B':[('G',1)],
    'C':[('G',1)],
    'G':[]
    }

h={
    'A':2,
    'B':1,
    'C':2,
    'G':0,
    }

def astar(start, goal):
    pq=[]
    heapq.heappush(pq, (h[start], 0, start, [start]))

    while pq:
        f,g,node,path = heapq.heappop(pq)

        if node == goal:
            print("A* Path:", path)
            print("Total Cost:", g)
            return
        for neighbor, cost in graph[node]:
            new_g = g + cost
            new_f = new_g + h[neighbor]
            heapq.heappush(pq,(new_f,new_g,neighbor,path + [neighbor]))
astar('A','G')

Aim: To implement the Apriori Algorithm for Association Rule Mining, find frequent itemsets, generate association rules, and calculate Support and Confidence.

Code:

import pandas as pd
from mlxtend.preprocessing import TransactionEncoder
from mlxtend.frequent_patterns import apriori, association_rules

transactions = [
['Bread', 'Milk', 'Butter'],
['Bread', 'Milk'],
['Bread', 'Butter'],
['Milk', 'Butter'],
['Bread', 'Milk', 'Butter']
]

te = TransactionEncoder()
te_array = te.fit(transactions).transform(transactions)
df = pd.DataFrame(te_array, columns=te.columns_)
print("Dataset")
print(df)

frequent_itemsets = apriori(df, min_support=0.4, use_colnames=True)
print("\nFrequent Itemsets")
print(frequent_itemsets)

rules = association_rules(frequent_itemsets,

metric="confidence",
min_threshold=0.6)
print("\nAssociation Rules")
print(rules[['antecedents',
'consequents',
'support',
'confidence']])
