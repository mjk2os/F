package org.example;
import io.github.bonigarcia.wdm.WebDriverManager;
import org.openqa.selenium.*;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.Select;
import org.openqa.selenium.support.ui.WebDriverWait;
import java.time.Duration;

public class WebElementDemo {

    public static void main(String[] args) {
        WebDriverManager.chromedriver().setup();
        WebDriver driver = new ChromeDriver();

        // Implicit Wait
        driver.manage().timeouts().implicitlyWait(Duration.ofSeconds(10));
        driver.manage().window().maximize();

        // Open Demo Website
        driver.get("https://demoqa.com/automation-practice-form");
        WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(10));

        // Textbox
        driver.findElement(By.id("firstName")).sendKeys("Manav");
        driver.findElement(By.id("lastName")).sendKeys("Pawar");
        driver.findElement(By.id("userEmail")).sendKeys("manavpawar287@gmail.com");

        // Radio Button
        WebElement male =
                driver.findElement(By.xpath("//label[text()='male']"));
        wait.until(ExpectedConditions.elementToBeClickable(male));
        male.click();

        // Mobile Number
        driver.findElement(By.id("userNumber"))
                .sendKeys("8600392378");

        // Checkbox
        WebElement sports =
                driver.findElement(By.xpath("//label[text()='Sports']"));
        wait.until(ExpectedConditions.elementToBeClickable(sports));
        sports.click();

        // Dropdown
        WebElement state =
                driver.findElement(By.id("react-select-3-input"));
        state.sendKeys("NCR");
        state.sendKeys(Keys.ENTER);

        // Explicit Wait Example
        wait.until(ExpectedConditions.visibilityOfElementLocated(
                By.id("submit")));
        driver.findElement(By.id("submit")).click();
        System.out.println("Form Submitted Successfully");
        driver.quit();
    }
}

StudentDataDriven.java:

package org.example;

import io.github.bonigarcia.wdm.WebDriverManager;
import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;
import org.openqa.selenium.*;
import org.openqa.selenium.chrome.ChromeDriver;
import java.io.*;

public class StudentDataDriven {
    public static void main(String[] args) throws Exception {
        Workbook wb = new XSSFWorkbook(new FileInputStream("students.xlsx"));
        Sheet sheet = wb.getSheetAt(0);
        WebDriverManager.chromedriver().setup();
        WebDriver driver = new ChromeDriver();
        String path = new File("student.html").getAbsolutePath();
        driver.get("file:///" + path.replace("\\", "/"));
        for (int i = 1; i <= 5; i++) {
            Row row = sheet.getRow(i);
            driver.findElement(By.id("roll")).clear();
            driver.findElement(By.id("name")).clear();
            driver.findElement(By.id("email")).clear();
            driver.findElement(By.id("roll")).sendKeys(row.getCell(0).toString());
            driver.findElement(By.id("name")).sendKeys(row.getCell(1).toString());
            driver.findElement(By.id("email")).sendKeys(row.getCell(2).toString());
            driver.findElement(By.id("submit")).click();
            String msg = driver.findElement(By.id("msg")).getText();
            row.createCell(3).setCellValue(msg.equals("Record Updated") ? "PASS" : "FAIL");
        }
        FileOutputStream out = new FileOutputStream("students.xlsx");
        wb.write(out);
        out.close();
        wb.close();
        driver.quit();
        System.out.println("Data Updated Successfully.");
    }
}

package org.example;
import org.openqa.selenium.*;
import org.openqa.selenium.chrome.ChromeDriver;
import java.io.File;
import java.util.List;

public class WebTableAnalysis {
    public static void main(String[] args) {
        WebDriver driver = new ChromeDriver();
        driver.get("file:///C:/Users/student/IdeaProjects/Practical8/studentMarks.html");
        List<WebElement> rows =
                driver.findElements(By.xpath("//table[@id='marks']//tr"));
        int count = 0;
        for (int i = 1; i < rows.size(); i++) {

            List<WebElement> cells =
                    rows.get(i).findElements(By.tagName("td"));
            int math = Integer.parseInt(cells.get(1).getText());
            int science = Integer.parseInt(cells.get(2).getText());
            int english = Integer.parseInt(cells.get(3).getText());
            if (math > 60 || science > 60 || english > 60) {
                count++;
            }
        }

        int total = rows.size() - 1;
        double percentage = (count * 100.0) / total;
        System.out.println("Total Students : " + total);
        System.out.println("Above 60 in at least one subject : " + count);
        System.out.printf("Percentage : %.2f%%", percentage);
        System.out.println("Rows found = " + rows.size());

    }
}

CheckboxValidation.java:

package org.example;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;

import java.util.List;

public class CheckboxValidation {

    public static void main(String[] args) {

        WebDriver driver = new ChromeDriver();

        driver.get("https://the-internet.herokuapp.com/checkboxes");
        driver.manage().window().maximize();

        List<WebElement> checkboxes =
                driver.findElements(By.cssSelector("input[type='checkbox']"));

        int checked = 0;
        int unchecked = 0;

        for (WebElement checkbox : checkboxes) {

            if (checkbox.isSelected())
                checked++;
            else
                unchecked++;
        }

        System.out.println("Total Checkboxes : " + checkboxes.size());
        System.out.println("Checked          : " + checked);
        System.out.println("Unchecked        : " + unchecked);

        System.out.println("\nSelecting all unchecked checkboxes...");

        for (WebElement checkbox : checkboxes) {

            if (!checkbox.isSelected()) {
                checkbox.click();
            }
        }
        boolean allSelected = true;

        for (WebElement checkbox : checkboxes) {

            if (!checkbox.isSelected()) {
                allSelected = false;
                break;
            }
        }

        if (allSelected)
            System.out.println("Validation Passed: All checkboxes are selected.");
        else
            System.out.println("Validation Failed.");

        driver.quit();
    }
}

package org.example;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class WebTableHandling {
    public static void main(String[] args) throws InterruptedException {
        WebDriver driver = new ChromeDriver();
        driver.get("https://the-internet.herokuapp.com/tables");
        driver.manage().window().maximize();

        List<WebElement> rows = driver.findElements(
                By.xpath("//table[@id='table1']/tbody/tr/td[1]"));

        System.out.println("Last Names in Table:");

        List<String> names = new ArrayList<>();

        for (WebElement row : rows) {
            System.out.println(row.getText());
            names.add(row.getText());
        }

        driver.findElement(
                        By.xpath("//table[@id='table1']//span[text()='Last Name']"))
                .click();

        List<WebElement> sortedRows = driver.findElements(
                By.xpath("//table[@id='table1']/tbody/tr/td[1]"));

        List<String> actual = new ArrayList<>();

        for (WebElement row : sortedRows) {
            actual.add(row.getText());
        }

        List<String> expected = new ArrayList<>(actual);
        Collections.sort(expected);

        if (actual.equals(expected))
            System.out.println("\nSorting Validation : Passed");
        else
            System.out.println("\nSorting Validation : Failed");

        System.out.println("\nNames containing 'B':");

        for (String name : actual) {
            if (name.startsWith("B")) {
                System.out.println(name);
            }
        }
        Thread.sleep(5000);
        driver.quit();
    }
}

public class ScreenshotLogDemo {

    static Logger log =
            LogManager.getLogger(ScreenshotLogDemo.class);
    public static void main(String[] args) {

        WebDriver driver = new ChromeDriver();
        driver.manage().window().maximize();
        driver.manage().timeouts()
                .implicitlyWait(Duration.ofSeconds(10));

        try {
            log.info("Browser Started");
            driver.get("https://www.google.com");
            log.info("Google Opened");

            // Intentionally incorrect locator to generate an error
            driver.findElement(By.id("wrong_id"));
            log.info("Element Found");
        }

        catch(Exception e){
            log.error("Test Failed");
            log.error(e.getMessage());
            takeScreenshot(driver);
        }

        finally {
            driver.quit();
            log.info("Browser Closed");
        }
    }

    public static void takeScreenshot(WebDriver driver){
        try{
            TakesScreenshot ts =
                    (TakesScreenshot) driver;
            File source =
                    ts.getScreenshotAs(OutputType.FILE);

//CREATE THE FOLDER IN SYSTEM OR GIVE THE PATH TO WHICH SCREENSHOT TO BE SAVED
            File destination =
                    new File("D:\\STQA\\screenshots\\Pract14.png");

            FileUtils.copyFile(source,destination);
            log.info("Screenshot Saved");
        }
        catch(Exception ex){
            log.error(ex.getMessage());
        }
    }
}
